Database schema + automated Blackout Radar detection queries & live inserts:

-- QuakeGrid Geospatial SQL Schema (Subang Jaya / Klang Valley)
CREATE TABLE Emergency_Reports (
    Report_ID INT PRIMARY KEY,
    Disaster_Type VARCHAR(50),
    Resource_Needed VARCHAR(100),
    Critical_Level VARCHAR(20),
    Latitude DECIMAL(10, 6),
    Longitude DECIMAL(10, 6),
    Status VARCHAR(20) DEFAULT 'Pending'
);

CREATE TABLE Donations (
    Donation_ID INT PRIMARY KEY,
    Item_Category VARCHAR(50),
    Quantity INT,
    Unit VARCHAR(20),
    Donor_Name VARCHAR(50),
    Dropoff_Location VARCHAR(100),
    Status VARCHAR(20) DEFAULT 'Pledged'
);

-- TABLE 3: Passive Device Heartbeats (Last-Ping Blackout Radar)
CREATE TABLE User_Heartbeats (
    Device_ID VARCHAR(50) PRIMARY KEY,
    Last_Lat DECIMAL(10, 6),
    Last_Lng DECIMAL(10, 6),
    Battery_Pct INT,
    Last_Ping TIMESTAMP,
    Connection_Status VARCHAR(20) DEFAULT 'Online' -- 'Online' | 'Signal Lost'
);

-- Automated Seismic Blackout Clustering Query:
SELECT Device_ID, Last_Lat, Last_Lng, Battery_Pct, Last_Ping
FROM User_Heartbeats
WHERE Connection_Status = 'Signal Lost'
  AND Last_Ping >= NOW() - INTERVAL 10 MINUTE;
